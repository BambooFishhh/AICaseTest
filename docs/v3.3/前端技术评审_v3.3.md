`streamGenerate` 新增 `cancelled` 事件监听 + `onCancelled` 回调：

```js
export function streamGenerate(projectId, { onProgress, onCase, onComplete, onCancelled, onError } = {}) {
  const url = `/api/projects/${projectId}/testcases/generate-stream`
  const es = new EventSource(url)

  es.addEventListener('progress', (e) => {
    try { onProgress?.(JSON.parse(e.data).message) } catch {}
  })
  es.addEventListener('case', (e) => {
    try { onCase?.(JSON.parse(e.data).testCase) } catch {}
  })
  es.addEventListener('complete', (e) => {
    try { onComplete?.(JSON.parse(e.data).total) } catch {}
    es.close()
  })
  // v3.3: 取消事件（区别于 error）
  es.addEventListener('cancelled', (e) => {
    try { onCancelled?.(JSON.parse(e.data).message) } catch {}
    es.close()
  })
  es.addEventListener('error', (e) => {
    let msg = '生成连接异常'
    if (e.data) {
      try { msg = JSON.parse(e.data).message || msg } catch {}
    }
    onError?.(msg)
    es.close()
  })

  return es
}
```

**脚本**：新增状态 + handleCancelGenerate + onCancelled 回调：

```js
import { cancelGenerate, streamGenerate, /* ... */ } from '@/api/testcase'

const cancelling = ref(false)  // v3.3: 取消中状态

// v3.3: 取消生成
async function handleCancelGenerate() {
  try {
    await ElMessageBox.confirm(
      '确定要取消生成吗？已生成的用例将被丢弃，旧用例会保留。',
      '确认取消',
      { confirmButtonText: '确定取消', cancelButtonText: '继续生成', type: 'warning' }
    )
  } catch {
    return
  }
  cancelling.value = true
  try {
    await cancelGenerate(projectId)
    // 后端会在下个检查点停止并推送 cancelled 事件
    // onCancelled 回调会处理状态清理
  } catch {
    ElMessage.error('取消请求失败')
    cancelling.value = false
  }
}

// handleRegenerate 的 streamGenerate 回调新增 onCancelled：
streamEs = streamGenerate(projectId, {
  onProgress: (msg) => { streamProgress.value = msg },
  onCase: (tc) => { streamedCases.value.unshift(tc) },
  onComplete: async (total) => {
    ElMessage.success(`用例生成完成，共 ${total} 条`)
    streaming.value = false
    streamProgress.value = ''
    regenerating.value = false
    page.value = 1
    await Promise.all([loadList(), loadAllForStats(), loadCoverageMatrix()])
  },
  onCancelled: async (msg) => {                     // ← v3.3 新增
    ElMessage.warning(msg || '生成已取消，旧用例已保留')
    streaming.value = false
    streamProgress.value = ''
    regenerating.value = false
    cancelling.value = false
    // 刷新列表（显示旧用例）
    await Promise.all([loadList(), loadAllForStats(), loadCoverageMatrix()])
  },
  onError: (msg) => {
    streaming.value = false
    streamProgress.value = ''
    regenerating.value = false
    cancelling.value = false
    generationError.value = msg
    ElMessage.error('用例生成失败')
  }
})
```

## 三、数据流

```
用户点击"取消生成"
  → ElMessageBox.confirm 确认
  → cancelGenerate(projectId) POST /generate-cancel
  → 后端 cancelGeneration 置 cancelled=true
  → 后端生成线程下个检查点抛 GenerationCancelledException
  → 后端 runGenerateStream catch: 跳过落库 + 恢复状态 + 推送 cancelled SSE 事件
  → 前端 EventSource 'cancelled' 事件 → onCancelled 回调
  → 前端: streaming=false + ElMessage.warning + 刷新列表(旧用例)
```

## 四、向后兼容

- 不传 `onCancelled` 回调时，cancelled 事件仅触发 `es.close()`，不报错
- 原有 progress/case/complete/error 事件处理不变
- `cancelGenerate` 对无生成任务的项目返回 cancelled:false，前端可忽略

## 五、测试验证

1. `npm run build` 构建通过
2. 流式生成期间点击"取消生成" → 确认 → 生成停止、显示黄色提示、旧用例保留
3. 流式生成期间关闭页面 → 后端日志显示取消、DB 旧用例保留
4. 正常生成完成 → 不出现取消按钮（streaming=false）
5. 取消按钮 loading 状态正确（cancelling=true 期间）
