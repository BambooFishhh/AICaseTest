## 一、改动概述

## 二、变更点详解

### 2.1 新建 playwright-mcp-server/index.js

独立 MCP Server，基于 Playwright 实现 9 个工具。

**核心设计**：
- 单浏览器单 tab 模型（不需要 tab 管理，简化实现）
- 全局 browser/context/page 状态
- video 录制在 `browser_launch` 时通过 `recordVideo` 选项启动

```javascript
import { chromium } from 'playwright';

let browser = null, context = null, page = null;

// browser_launch 工具
case 'browser_launch': {
  browser = await chromium.launch({ headless: args.headless ?? true });
  const ctxOpts = { viewport: { width: args.width ?? 1280, height: args.height ?? 720 } };
  if (args.video_dir) {
    fs.mkdirSync(args.video_dir, { recursive: true });
    ctxOpts.recordVideo = { dir: args.video_dir, size: { width: 1280, height: 720 } };
  }
  context = await browser.newContext(ctxOpts);
  page = await context.newPage();
  return { content: [{ type: 'text', text: 'ok' }] };
}

// browser_take_screenshot 工具
case 'browser_take_screenshot': {
  const buf = await page.screenshot();
  const filePath = args.path;
  fs.writeFileSync(filePath, buf);
  return { content: [{ type: 'text', text: filePath }] };
}

// browser_visual_click 工具
case 'browser_visual_click': {
  await page.mouse.click(args.x, args.y);
  return { content: [{ type: 'text', text: `clicked (${args.x},${args.y})` }] };
}

// browser_video_save 工具
case 'browser_video_save': {
  const video = page.video();
  if (video) { await video.saveAs(args.filename); return { content: [{ type: 'text', text: args.filename }] }; }
  return { content: [{ type: 'text', text: 'no video' }] };
}
```

### 2.2 新建 PlaywrightRecordSkill.java

方法签名与 BrowserSkill 对齐，内部调用 playwright MCP Server。

**关键设计**：
- `sessionId` 参数保留但忽略（Playwright 单浏览器模型不需要）
- `takeScreenshotWithMarker` 复用 BrowserSkill 的 Graphics2D 标注逻辑
- `startRecording` 无操作（launch 时已通过 video_dir 启动录屏）
- `stopRecording` 调用 `browser_video_save` 返回 mp4 路径（不再是图片列表）

```java
@Component
public class PlaywrightRecordSkill {

    @Autowired
    private McpClientManager mcpClientManager;

    @Value("${app.output-dir:outputs}")
    private String outputDir;

    public String browserLaunch(boolean headless, int width, int height, String videoDir) {
        String dir = videoDir != null ? videoDir : outputDir + "/recordings";
        mcpClientManager.callTool("playwright", "browser_launch", Map.of(
            "headless", headless, "width", width, "height", height, "video_dir", dir
        ));
        return "playwright-session";  // 固定 sessionId
    }

    public String takeScreenshot(String sessionId) {
        String path = outputDir + "/screenshots/" + System.currentTimeMillis() + ".png";
        return mcpClientManager.callTool("playwright", "browser_take_screenshot",
            Map.of("path", path));
    }

    public String takeScreenshotWithMarker(String sessionId, int clickX, int clickY) {
        String path = takeScreenshot(sessionId);
        if (clickX > 0 && clickY > 0) {
            annotateScreenshot(path, clickX, clickY);  // 复用 Graphics2D 标注
        }
        return path;
    }

    public String stopRecording(String filename) {
        return mcpClientManager.callTool("playwright", "browser_video_save",
            Map.of("filename", filename));
    }
}
```

@PostConstruct
public void start() {
