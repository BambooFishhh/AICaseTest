### 1.1 CoverageMatrix.vue — 覆盖率矩阵组件（F2）

文件：`frontend/src/components/CoverageMatrix.vue`（新增）

```html
<template>
  <el-card class="coverage-matrix">
    <template #header>
      <span>覆盖率矩阵</span>
      <el-tag v-if="matrix" type="info" size="small" style="margin-left: 8px">
        {{ matrix.summary.coveredTransitions }} / {{ matrix.summary.totalTransitions }}
        ({{ Math.round(matrix.summary.rate * 100) }}%)
      </el-tag>
    </template>

    <el-table
      v-if="allTransitions.length > 0"
      :data="allTransitions"
      border
      size="small"
      :row-class-name="rowClassName"
      style="width: 100%"
    >
      <el-table-column prop="smName" label="状态机" width="140" />
      <el-table-column prop="from" label="From" width="120" />
      <el-table-column prop="to" label="To" width="120" />
      <el-table-column prop="trigger" label="Trigger" width="120" />
      <el-table-column label="覆盖" width="80" align="center">
        <template #default="{ row }">
          <el-tag :type="row.covered ? 'success' : 'danger'" size="small">
            {{ row.covered ? '✓ 已覆盖' : '✗ 未覆盖' }}
          </el-tag>
        </template>
      </el-table-column>
      <el-table-column label="关联用例" width="100" align="center">
        <template #default="{ row }">
          <el-button
            v-if="row.testCaseIds.length > 0"
            type="primary"
            link
            size="small"
            @click="$emit('filter-by-ids', row.testCaseIds)"
          >
            {{ row.testCaseIds.length }} 条
          </el-button>
          <span v-else>—</span>
        </template>
      </el-table-column>
    </el-table>
    <el-empty v-else description="暂无状态机数据" />
  </el-card>
</template>

<script setup>
import { computed } from 'vue'

const props = defineProps({
  matrix: { type: Object, default: null }
})

defineEmits(['filter-by-ids'])

const allTransitions = computed(() => {
  if (!props.matrix || !props.matrix.stateMachines) return []
  const result = []
  for (const sm of props.matrix.stateMachines) {
    for (const tran of (sm.transitions || [])) {
      result.push({
        ...tran,
        smName: sm.name,
        smId: sm.id
      })
    }
  }
  return result
})

const rowClassName = ({ row }) => {
  return row.covered ? '' : 'row-uncovered'
}
</script>

<style scoped>
.coverage-matrix {
  margin-top: 16px;
}
:deep(.row-uncovered) {
  background-color: #fef0f0;
}
</style>
```

### 1.4 StateMachineViewer.vue — 覆盖标注增强

文件：`frontend/src/components/StateMachineViewer.vue`

新增 `coverageData` prop，用于标注边和节点的覆盖状态：

```js
const props = defineProps({
  states: { type: Array, default: () => [] },
  transitions: { type: Array, default: () => [] },
  forbiddenTransitions: { type: Array, default: () => [] },
  coverageData: { type: Object, default: null }  // v1.5 新增
})
```

`buildEdges()` 中根据 coverageData 标注边颜色：

```js
const isCovered = (tran) => {
  if (!props.coverageData) return null  // 无覆盖数据时不标注
  const sm = props.coverageData.stateMachines?.[0]
  if (!sm) return null
  const match = (sm.transitions || []).find(t =>
    t.from === tran.from && t.to === tran.to
  )
  return match ? match.covered : null
}

// 在 buildEdges 中:
const covered = isCovered(tran)
return {
  ...,
  lineStyle: {
    color: covered === false ? '#F56C6C' : (covered === true ? '#67C23A' : '#909399'),
    width: covered === false ? 2 : 2,
    type: covered === false ? 'dashed' : 'solid',
    curveness: 0.2
  }
}
```
