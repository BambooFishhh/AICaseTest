export function exportTestCases(projectId, format, ids) {
  return request.get(`/projects/${projectId}/testcases/export`, {
    params: { format, ids: ids && ids.length ? ids.join(',') : undefined },
    responseType: 'blob'
  })
}

// v1.7: 导入 JSON 文件
export function importTestCases(projectId, file) {
  const form = new FormData()
  form.append('file', file)
  return request.post(`/projects/${projectId}/testcases/import`, form, {
    headers: { 'Content-Type': 'multipart/form-data' }
  })
}

// v1.7: 跨项目复制
export function copyToProject(projectId, ids, targetProjectId) {
  return request.post(`/projects/${projectId}/testcases/copy-to`, {
    ids,
    targetProjectId
  })
}

### 1.3 script 新增方法

```js
import { exportTestCases, importTestCases, copyToProject } from '@/api/testcase'
import { listProjects } from '@/api/project'
import { Upload, CopyDocument, Document } from '@element-plus/icons-vue'
import { saveAs } from '@/utils/file'  // 或自行实现 blob 下载

const importFileInput = ref(null)

async function downloadBlob(res, defaultName) {
  const disposition = res.headers?.['content-disposition'] || ''
  const match = disposition.match(/filename="?([^"]+)"?/)
  const fileName = match ? match[1] : defaultName
  const blob = new Blob([res.data])
  const url = URL.createObjectURL(blob)
  const a = document.createElement('a')
  a.href = url
  a.download = decodeURIComponent(fileName)
  a.click()
  URL.revokeObjectURL(url)
}

async function handleExportJson() {
  const ids = selectedRows.value.map((tc) => tc.id)
  try {
    const res = await exportTestCases(projectId, 'json', ids)
    downloadBlob(res, 'testcases.json')
    ElMessage.success('JSON 导出成功')
  } catch {}
}

async function handleExportCsv() {
  try {
    const res = await exportTestCases(projectId, 'csv', null)
    downloadBlob(res, 'testcases.csv')
    ElMessage.success('CSV 导出成功')
  } catch {}
}

function triggerImportFile() {
  importFileInput.value?.click()
}

async function handleImportFile(e) {
  const file = e.target.files?.[0]
  if (!file) return
  try {
    const res = await importTestCases(projectId, file)
    ElMessage.success(`导入完成：成功 ${res.data.imported} 条，跳过 ${res.data.skipped} 条`)
    await Promise.all([loadList(), loadAllForStats(), loadCoverageMatrix()])
  } catch {}
  e.target.value = ''  // 重置以便重复导入同一文件
}

async function handleCopyTo() {
  const ids = selectedRows.value.map((tc) => tc.id)
  try {
    const projRes = await listProjects()
    const projects = (projRes.data || []).filter((p) => p.id !== projectId)
    if (projects.length === 0) {
      ElMessage.warning('没有可复制的目标项目')
      return
    }
    const { value: targetId } = await ElMessageBox.prompt(
      '选择目标项目：输入项目 ID',
      '复制用例到其他项目',
      {
        inputType: 'text',
        inputValidator: (v) => !!v || '请输入目标项目 ID',
        inputValue: projects[0].id
      }
    )
    const res = await copyToProject(projectId, ids, targetId)
    ElMessage.success(`已复制 ${res.data.copied} 条用例到目标项目`)
  } catch {}
}
```

### 1.4 说明
- listProjects 已存在于 `@/api/project`，复用
- blob 下载用原生 `URL.createObjectURL` + `<a download>`，无需第三方库
- CSV 导出按钮不依赖选中行（导出全部）
- 导入成功后刷新列表/统计/覆盖率矩阵
- 文件 input 用 `display:none` + 程序触发 click，避免原生 input 样式问题
