### 2.2 新增导入 XMind

模板：
```html
<el-button :icon="Upload" @click="triggerImportXmind">导入XMind</el-button>
<input ref="xmindFileInput" type="file" accept=".xmind" style="display: none" @change="handleImportXmind" />
```

Script：
```javascript
const xmindFileInput = ref(null)
const mindmapGenerated = ref(false) // v3.9: 标记脑图是否已生成

function triggerImportXmind() {
  xmindFileInput.value?.click()
}

async function handleImportXmind(e) {
  const file = e.target.files[0]
  if (!file) return
  try {
    const res = await importXmind(projectId, file)
    ElMessage.success(`导入完成：成功 ${res.data.imported} 条，跳过 ${res.data.skipped} 条`)
    loadList()
  } catch (err) {
    ElMessage.error(err.message || '导入失败')
  }
  e.target.value = ''
}
```

API：
```javascript
export function importXmind(projectId, file) {
  const formData = new FormData()
  formData.append('file', file)
  return request.post(`/projects/${projectId}/testcases/import-xmind`, formData, {
    headers: { 'Content-Type': 'multipart/form-data' },
  })
}
```

### 2.3 查看脑图入口

生成脑图成功后设置 `mindmapGenerated = true`，显示"查看脑图"按钮：

```html
<el-button v-if="mindmapGenerated" type="info" :icon="View" @click="handleViewMindmap">
  查看脑图
</el-button>
```

```javascript
function handleViewMindmap() {
  router.push(`/projects/${projectId}/mindmap`)
}

async function handleGenerateMindmap() {
  generatingMap.value = true
  try {
    await generateMindmap(projectId)
    ElMessage.success('脑图生成成功')
    mindmapGenerated.value = true  // v3.9: 显示查看按钮
  } finally {
    generatingMap.value = false
  }
}
```
