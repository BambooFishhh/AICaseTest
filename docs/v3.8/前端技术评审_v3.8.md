## 2. 改动详情

### 2.1 树状数据 computed

```javascript
// v3.8: 树状数据——按模块分组
const treeData = computed(() => {
  const cases = streaming.value
    ? (currentGenMode.value === 'append'
        ? [...streamedCases.value, ...testCases.value]
        : streamedCases.value)
    : testCases.value

  // priority 客户端筛选
  const filtered = filters.priority
    ? cases.filter(tc => tc.priority === filters.priority)
    : cases

  // 按模块分组
  const moduleMap = new Map()
  filtered.forEach(tc => {
    const mod = tc.module || '未分类'
    if (!moduleMap.has(mod)) moduleMap.set(mod, [])
    moduleMap.get(mod).push(tc)
  })

  const tree = []
  moduleMap.forEach((children, mod) => {
    tree.push({
      id: `module-${mod}`,
      isModule: true,
      title: `${mod} (${children.length}条)`,
      module: mod,
      children
    })
  })
  return tree
})
```

### 2.4 行点击 + 行样式

```javascript
function rowClassName({ row }) {
  return row.isModule ? 'module-row' : 'case-row'
}

function handleRowClick(row) {
  if (row.isModule) return
  currentTestCase.value = row
  dialogVisible.value = true
}
```
