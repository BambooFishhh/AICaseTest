### 2.1 MCP Server (`mcp-server/index.js`)

#### 2.1.1 StreamingServer 子类

MCP SDK 的 `Server` 类继承 `Protocol`，`notification()` 方法是 `protected`。创建子类暴露它：

```javascript
import { Server } from '@modelcontextprotocol/sdk/server/index.js';

class StreamingServer extends Server {
  async sendChunkNotification(text, index) {
    await this.notification({
      method: "notifications/llm_chunk",
      params: { text, index }
    });
  }
}

// 替换: const server = new Server(...)
const server = new StreamingServer(
  { name: 'aicasetest-mcp-server', version: '1.2.0' },
  { capabilities: { tools: {} } }
);
```

handler 改造：

```javascript
case 'llm_chat': {
  const { system_prompt, user_prompt, temperature = 0.7, stream = false } = args;
  if (stream) {
    let fullText = '';
    const completion = await client.chat.completions.create({
      model,
      temperature: parseFloat(temperature),
      max_tokens: 8192,
      stream: true,
      messages: [
        { role: 'system', content: system_prompt },
        { role: 'user', content: user_prompt },
      ],
    });
    let chunkIndex = 0;
    for await (const chunk of completion) {
      const delta = chunk.choices[0]?.delta?.content || '';
      if (delta) {
        fullText += delta;
        await server.sendChunkNotification(delta, chunkIndex++);
      }
    }
    return { content: [{ type: 'text', text: fullText }] };
  }
  // 非流式：原有逻辑不变
  const response = await client.chat.completions.create({
    model,
    temperature: parseFloat(temperature),
    max_tokens: 8192,
    messages: [
      { role: 'system', content: system_prompt },
      { role: 'user', content: user_prompt },
    ],
  });
  const text = response.choices[0]?.message?.content || '';
  return { content: [{ type: 'text', text }] };
}
```

### 2.2 McpConnection.callToolStreaming

新增 `callToolStreaming` 方法，与 `callTool` 对称但循环读取：

```java
/**
 * v3.7: 流式调用工具。逐块读取 stdout，dispatch llm_chunk 通知到 chunkConsumer，
 * 等待匹配 id 的 JSON-RPC response 后返回。
 */
public synchronized String callToolStreaming(
        String toolName, Map<String, Object> args,
        java.util.function.Consumer<String> chunkConsumer) throws Exception {

    if (!initialized || process == null || !process.isAlive()) {
        log.warn("MCP [{}] 未启动或已死, 尝试重启...", name);
        start();
        if (!initialized) {
            throw new IllegalStateException("MCP Server [" + name + "] 未启动");
        }
    }

    int id = requestId.incrementAndGet();
    ObjectNode request = objectMapper.createObjectNode();
    request.put("jsonrpc", "2.0");
    request.put("id", id);
    request.put("method", "tools/call");
    ObjectNode params = objectMapper.createObjectNode();
    params.put("name", toolName);
    params.set("arguments", objectMapper.valueToTree(args));
    request.set("params", params);

    String json = objectMapper.writeValueAsString(request);
    log.info("MCP [{}] → callToolStreaming: {}, id={}, stream=true", name, toolName, id);
    stdin.write(json);
    stdin.newLine();
    stdin.flush();

    // 循环读取 stdout 行
    while (true) {
        String line = stdout.readLine();
        if (line == null) {
            throw new IOException("MCP Server [" + name + "] stdout 已关闭");
        }

        JsonNode msg = objectMapper.readTree(line);

        // notification: 有 method 无 id → dispatch
        if (msg.has("method") && !msg.has("id")) {
            String method = msg.path("method").asText();
            if ("notifications/llm_chunk".equals(method) && chunkConsumer != null) {
                String chunkText = msg.path("params").path("text").asText("");
                if (!chunkText.isEmpty()) {
                    chunkConsumer.accept(chunkText);
                }
            }
            continue;
        }

        // response: 有 id → 匹配则返回
        if (msg.has("id") && msg.path("id").asInt() == id) {
            if (msg.has("error")) {
                throw new RuntimeException("MCP [" + name + "] 错误: " + msg.get("error"));
            }
            JsonNode content = msg.path("result").path("content");
            if (content.isArray() && !content.isEmpty()) {
                String text = content.get(0).path("text").asText("");
                log.info("MCP [{}] callToolStreaming 完成, id={}, 长度={}", name, id, text.length());
                return text;
            }
            throw new RuntimeException("MCP Server [" + name + "] 返回内容为空");
        }
        // id 不匹配 → 忽略（不应在 synchronized 模式下发生）
    }
}
```

### 2.4 LlmService.chatStreaming

```java
/**
 * v3.7: 流式调用 LLM。chunkConsumer 接收逐块文本，方法返回完整响应。
 */
public String chatStreaming(String systemPrompt, String userPrompt,
                           double temperature,
                           java.util.function.Consumer<String> chunkConsumer) {
    log.info("[LLM] chatStreaming() 开始, prompt长度={}", userPrompt == null ? 0 : userPrompt.length());
    Exception lastException = null;
    for (int attempt = 0; attempt < MAX_RETRIES; attempt++) {
        try {
            long start = System.currentTimeMillis();
            Map<String, Object> args = new LinkedHashMap<>();
            args.put("system_prompt", systemPrompt);
            args.put("user_prompt", userPrompt);
            args.put("temperature", temperature);
            args.put("stream", true);

            String result = mcpClientManager.callToolStreaming("llm", "llm_chat", args, chunkConsumer);
            long elapsed = System.currentTimeMillis() - start;
            log.info("[LLM] chatStreaming 完成, 耗时={}ms, 响应长度={}", elapsed, result == null ? 0 : result.length());
            return result;
        } catch (Exception e) {
            lastException = e;
            log.warn("[LLM] chatStreaming attempt {} 失败: {}", attempt + 1, e.getMessage());
        }
        if (attempt < MAX_RETRIES - 1) {
            try { Thread.sleep(RETRY_DELAYS_MS[attempt]); } catch (InterruptedException ie) {
                Thread.currentThread().interrupt(); break;
            }
        }
    }
    throw new BusinessException(50002,
            "LLM流式调用失败（已重试" + MAX_RETRIES + "次）: " + (lastException != null ? lastException.getMessage() : "unknown"),
            HttpStatus.INTERNAL_SERVER_ERROR);
}
```

### 2.5 TestGeneratorAgent — StreamingTestCaseParser

新增内部类，增量解析 JSON 数组：

```java
/**
 * v3.7: 流式 JSON 数组解析器。
 * 积累文本 chunk，检测完整用例对象后立即回调 caseCb。
 */
public class StreamingTestCaseParser {
    private final StringBuilder buffer = new StringBuilder();
    private final CaseCallback caseCb;
    private final ObjectMapper mapper;

    private int scanPos = 0;          // 已扫描到的位置
    private int arrayStart = -1;      // 数组 [ 的位置
    private int objStart = -1;        // 当前对象 { 的位置
    private int braceDepth = 0;       // 花括号深度
    private boolean inString = false;  // 是否在字符串内
    private boolean escaped = false;   // 前一个字符是否为转义符
    private int parsedCount = 0;      // 已解析的用例数

    public StreamingTestCaseParser(CaseCallback caseCb, ObjectMapper mapper) {
        this.caseCb = caseCb;
        this.mapper = mapper;
    }

    public void append(String chunk) {
        buffer.append(chunk);
        scan();
    }

    private void scan() {
        String text = buffer.toString();
        int len = text.length();
        for (int i = scanPos; i < len; i++) {
            char c = text.charAt(i);

            if (inString) {
                if (escaped) {
                    escaped = false;
                } else if (c == '\\') {
                    escaped = true;
                } else if (c == '"') {
                    inString = false;
                }
                continue;
            }

            // 不在字符串内
            if (arrayStart == -1) {
                // 寻找数组开始
                if (c == '[') {
                    arrayStart = i;
                }
                continue;
            }

            // 在数组内
            if (c == '"') {
                inString = true;
            } else if (c == '{') {
                if (braceDepth == 0) {
                    objStart = i;
                }
                braceDepth++;
            } else if (c == '}') {
                braceDepth--;
                if (braceDepth == 0 && objStart >= 0) {
                    // 完整对象！提取并解析
                    String objJson = text.substring(objStart, i + 1);
                    try {
                        parseAndCallback(objJson);
                    } catch (Exception e) {
                        log.warn("流式解析用例对象失败, 跳过: {}", e.getMessage());
                    }
                    objStart = -1;
                }
            } else if (c == ']' && braceDepth == 0) {
                // 数组结束
                break;
            }
        }
        scanPos = len;
    }

    private void parseAndCallback(String objJson) throws Exception {
        JsonNode node = mapper.readTree(objJson);
        TestCase tc = new TestCase();
        tc.setTitle(node.path("title").asText("未命名测试用例"));
        tc.setModule(node.path("module").asText("未分类"));
        tc.setType(node.path("type").asText("positive"));
        tc.setPriority(node.path("priority").asText("P1"));
        tc.setPreconditions(serializeStringArray(node.path("preconditions")));
        tc.setSteps(serializeStringArray(node.path("steps")));
        tc.setExpectedResults(serializeStringArray(node.path("expectedResults")));
        tc.setStructuredSteps(nodeToJson(node.path("structuredSteps"), "[]"));
        tc.setApiEndpoints(nodeToJson(node.path("apiEndpoints"), "[]"));
        tc.setTestData(nodeToJson(node.path("testData"), "{}"));
        tc.setExecutionHints(nodeToJson(node.path("executionHints"), "{}"));
        tc.setStateMachineRef(nodeToJson(node.path("stateMachineRef"), "{}"));
        tc.setSource("ai_generation");
        tc.setConfidence(0.8);
        parsedCount++;
        if (caseCb != null) {
            try { caseCb.onCase(tc); } catch (Exception ex) {
                log.warn("caseCallback failed, continue: {}", ex.getMessage());
            }
        }
    }

    public int getParsedCount() { return parsedCount; }
}
```

```java
// generateByLlmWithPrd 改造:
private List<TestCase> generateByLlmWithPrd(PrdAnalysisResult prdResult, ...,
                                            CaseCallback caseCb, AtomicBoolean cancelled,
                                            GenerationParams params) throws Exception {
    // ... 构建 context/prompt（不变） ...

    checkCancelled(cancelled);
    
    // v3.7: 流式调用 + 增量解析
    if (caseCb != null) {
        StreamingTestCaseParser parser = new StreamingTestCaseParser(caseCb, objectMapper);
        String response = llmService.chatStreaming(
                buildPrdDrivenPrompt(params), userPrompt, resolveTemperature(params),
                parser::append);
        // 兜底：用完整响应重新解析，确保不遗漏
        String json = extractJsonArray(response);
        List<TestCase> all = parseTestCases(json, null); // caseCb=null 避免重复
        // 只推送流式期间未推送的（通过数量对比）
        int streamed = parser.getParsedCount();
        for (int i = streamed; i < all.size(); i++) {
            if (caseCb != null) {
                try { caseCb.onCase(all.get(i)); } catch (Exception ex) {
                    log.warn("兜底推送失败: {}", ex.getMessage());
                }
            }
        }
        return all;
    }
    
    // caseCb 为 null（非流式场景）：原有逻辑不变
    String response = llmService.chat(buildPrdDrivenPrompt(params), userPrompt, resolveTemperature(params));
    String json = extractJsonArray(response);
    return parseTestCases(json, null);
}
```
